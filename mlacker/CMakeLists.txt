cmake_minimum_required(VERSION 3.19)
project(mlacker VERSION 0.1.0 LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "mlacker currently requires macOS AUHAL")
endif()
if(NOT VST3_SDK_ROOT OR NOT MLACKER_MAIN_OBJECT)
    message(FATAL_ERROR "Use mlang pkg build, or specify VST3_SDK_ROOT and MLACKER_MAIN_OBJECT")
endif()
get_filename_component(VST3_SDK_ROOT "${VST3_SDK_ROOT}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(MLACKER_MAIN_OBJECT "${MLACKER_MAIN_OBJECT}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(MLANG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_ADD_VST3_UTILITIES OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_DISABLE_CODE_SIGNING ON CACHE BOOL "" FORCE)
set(SMTG_BUILD_UNIVERSAL_BINARY OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/sdk")
smtg_enable_vst3_sdk()
find_package(OpenSSL REQUIRED)
add_library(mlacker_host STATIC src/vst3_host.cpp
    "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp"
    "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/plugprovider.cpp"
    "${VST3_SDK_ROOT}/public.sdk/source/vst/hosting/module_mac.mm")
target_include_directories(mlacker_host PUBLIC "${MLANG_ROOT}/stdlib/include")
target_compile_options(mlacker_host PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
target_link_libraries(mlacker_host PUBLIC sdk_hosting "${MLANG_ROOT}/build/libmlang_std.a" OpenSSL::SSL OpenSSL::Crypto
    "-framework Cocoa" "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox"
    "-framework AudioUnit" "-framework CoreMIDI" "-framework AVFoundation" "-framework CoreGraphics" "-framework ImageIO")
set_source_files_properties("${MLACKER_MAIN_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(mlacker "${MLACKER_MAIN_OBJECT}")
set_target_properties(mlacker PROPERTIES LINKER_LANGUAGE CXX RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(mlacker PRIVATE mlacker_host)
if(EXISTS "${MLACKER_MAIN_OBJECT}.mlang-info.plist")
    target_link_options(mlacker PRIVATE "LINKER:-sectcreate,__TEXT,__info_plist,${MLACKER_MAIN_OBJECT}.mlang-info.plist")
endif()

enable_testing()
smtg_add_vst3plugin(MlackerTestInstrument tests/test_instrument.cpp
    "${VST3_SDK_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
target_link_libraries(MlackerTestInstrument PRIVATE sdk)
smtg_target_set_bundle(MlackerTestInstrument BUNDLE_IDENTIFIER org.mlang.mlacker.test COMPANY_NAME mlacker)
smtg_add_vst3plugin(MlackerTestEffect tests/test_instrument.cpp
    "${VST3_SDK_ROOT}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
target_compile_definitions(MlackerTestEffect PRIVATE MLACKER_TEST_EFFECT=1)
target_link_libraries(MlackerTestEffect PRIVATE sdk)
smtg_target_set_bundle(MlackerTestEffect BUNDLE_IDENTIFIER org.mlang.mlacker.testeffect COMPANY_NAME mlacker)
add_executable(mlacker_vst3_tests tests/vst3_tests.cpp)
target_link_libraries(mlacker_vst3_tests PRIVATE mlacker_host)
add_dependencies(mlacker_vst3_tests MlackerTestInstrument MlackerTestEffect)
add_test(NAME vst3_host COMMAND mlacker_vst3_tests "$<TARGET_BUNDLE_DIR:MlackerTestInstrument>" "$<TARGET_BUNDLE_DIR:MlackerTestEffect>")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
# Sampler-pad end to end runs when plugins/mla_drum has been built.
set(MLA_DRUM_VST3 "${MLANG_ROOT}/plugins/mla_drum/build/cmake/VST3/Release/MlaDrum.vst3")
set(MLA_DELAY_VST3 "${MLANG_ROOT}/plugins/mla_delay/build/cmake/VST3/Release/MlaDelay.vst3")
# Real bundles built in this tree join the host test when they exist.
set(VST3_HOST_TEST_ENV "")
if(EXISTS "${MLA_DELAY_VST3}")
    list(APPEND VST3_HOST_TEST_ENV "MLA_DELAY_VST3=${MLA_DELAY_VST3}")
endif()
if(EXISTS "${MLA_DRUM_VST3}")
    list(APPEND VST3_HOST_TEST_ENV "MLA_DRUM_VST3=${MLA_DRUM_VST3}")
endif()
if(VST3_HOST_TEST_ENV)
    set_tests_properties(vst3_host PROPERTIES ENVIRONMENT "${VST3_HOST_TEST_ENV}")
endif()
if(EXISTS "${MLA_DRUM_VST3}")
    add_test(NAME sampler_pads_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/sampler_pads_tui_smoke.py" "$<TARGET_FILE:mlacker>" "${MLA_DRUM_VST3}")
endif()

add_test(NAME vst3_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/vst3_tui_smoke.py" "$<TARGET_FILE:mlacker>" "$<TARGET_BUNDLE_DIR:MlackerTestInstrument>")
add_test(NAME session_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/session_tui_smoke.py" "$<TARGET_FILE:mlacker>" "$<TARGET_BUNDLE_DIR:MlackerTestInstrument>")
add_test(NAME vertical_zoom_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/vertical_zoom_tui_smoke.py" "$<TARGET_FILE:mlacker>")
set_tests_properties(vertical_zoom_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 60)
add_test(NAME record_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/record_tui_smoke.py" "$<TARGET_FILE:mlacker>")
set_tests_properties(record_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 60)
add_test(NAME patterns_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/patterns_tui_smoke.py" "$<TARGET_FILE:mlacker>")
set_tests_properties(patterns_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 60)
add_test(NAME effects_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/effects_tui_smoke.py" "$<TARGET_FILE:mlacker>" "$<TARGET_BUNDLE_DIR:MlackerTestInstrument>" "$<TARGET_BUNDLE_DIR:MlackerTestEffect>")
set_tests_properties(effects_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 120)
add_test(NAME inserts_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/inserts_tui_smoke.py" "$<TARGET_FILE:mlacker>" "$<TARGET_BUNDLE_DIR:MlackerTestEffect>")
set_tests_properties(inserts_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 120)

add_test(NAME matrix_record_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/matrix_record_tui_smoke.py" "$<TARGET_FILE:mlacker>")
set_tests_properties(matrix_record_tui PROPERTIES TIMEOUT 60)

set_tests_properties(output_routing_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 120)
add_test(NAME spectrum_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/spectrum_tui_smoke.py" "$<TARGET_FILE:mlacker>")
set_tests_properties(spectrum_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 60)
add_test(NAME visual_tui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/visual_tui_smoke.py" "$<TARGET_FILE:mlacker>")
set_tests_properties(visual_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 60)
set_tests_properties(session_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 120)
add_test(NAME terminal_tui COMMAND "${Python3_EXECUTABLE}" "${MLANG_ROOT}/tests/tui_terminal_smoke.py" "$<TARGET_FILE:mlacker>")
add_test(NAME playback_tui COMMAND "${Python3_EXECUTABLE}" "${MLANG_ROOT}/tests/tui_playback_smoke.py" "$<TARGET_FILE:mlacker>")
set_tests_properties(settings_tui vst3_tui terminal_tui playback_tui PROPERTIES WORKING_DIRECTORY "${MLANG_ROOT}" TIMEOUT 120)
set_tests_properties(settings_tui vst3_tui terminal_tui playback_tui PROPERTIES ENVIRONMENT "MLANG_TUI_DEMO=1")
set_tests_properties(vst3_host PROPERTIES TIMEOUT 30)
